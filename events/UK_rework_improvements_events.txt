# Selection of a Hohenzollern Monarch
country_event = {
	id = ukrework_improvements.4
	title = ukrework_improvements.4.t
	desc = ukrework_improvements.4.desc
	picture = GFX_report_event_royal_visit

	is_triggered_only = yes

	trigger = {
	}

	option = { # Approach the eldest son (Wilhelm III)
		ai_chance = {
			base = 5
		}
		name = ukrework_improvements.4.a
		trigger = {
			NOT = {
				GER = {
					OR = {
						has_country_leader = {
							character = GER_wilhelm_ii
							ruling_only = yes
						}
						has_country_leader = {
							character = GER_wilhelm_iii
							ruling_only = yes
						}
						has_idea = GER_wilhelm_iii
					}
				}
			}
		}
		custom_effect_tooltip = ukrework_improvements.4.a.tt
		custom_effect_tooltip = generic_skip_one_line_tt
			IF = {
				limit = {
					GER = {
						AND = {
							has_government = fascism
							has_war_with = FRA
							OR = {
								has_war_with = POL
								has_war_with = CZE
							}
						}
					}
				}
				random_list = {
					30 = { ENG_wilhelm_iii_joins_effect = yes } # Wilhelm agrees
					70 = { country_event = { id = ukrework_improvements.5 days = 1 } } # Wilhelm refuses
				}
			}
			ELSE = {
				random_list = {
					10 = { ENG_wilhelm_iii_joins_effect = yes } # Wilhelm agrees
					90 = { country_event = { id = ukrework_improvements.5 days = 1 } } # Wilhelm refuses
				}
			}
		#}
	}

	option = { # Don't bother him, we want a separate monarch!
		ai_chance = {
			base = 5
		}
		name = ukrework_improvements.4.b
		trigger = {
			NOT = {
				any_country = {
					OR = {
						has_country_leader = {
							character = GER_victoria
							ruling_only = yes
						}
						has_idea = GER_victoria_louise
						has_idea = victoria_ii
					}
				}
			}
		}
		custom_effect_tooltip = ukrework_improvements.4.b.tt
		hidden_effect = {
			ENG_victoria_joins_effect = yes
			set_country_flag = ENG_promised_unionjack_return
		}
	}
}

country_event = {
	id = ukrework_improvements.5
	title = ukrework_improvements.5.t
	desc = ukrework_improvements.5.desc
	picture = GFX_report_event_royal_visit

	is_triggered_only = yes
	trigger = {
	}
	option = { # ask his brother
		name = ukrework_improvements.5.a
		trigger = {
			NOT = {
				has_country_flag = PRE_eitel_friedrich_refused_throne
				any_country = {
					exists = yes
					has_country_leader = {
						character = PRE_eitel_friedrich
						ruling_only = yes
					}
				}
				PRE = {
					exists = yes
				}
			}
		}
		random_list = {
			25 = { ENG_eitel_friedrich_joins_effect = yes } # Eitel Friedrich agrees
			75 = { set_country_flag = PRE_eitel_friedrich_refused_throne country_event = { id = ukrework_improvements.5 days = 1 } } # Eitel Friedrich refuses
		}
	}
	option = { # ask his son
		name = ukrework_improvements.5.b
		trigger = {
			NOT = {
				has_country_flag = ENG_wilhelm_iv_refused_throne
				any_country = {
					exists = yes
					has_country_leader = {
						character = GER_wilhelm_iv
						ruling_only = yes
					}
				}
				GER = { # Westfeldzug has begun -> Wilhelm IV died there
					has_government = fascism
					has_war_with = FRA
					OR = {
						has_war_with = HOL
						has_war_with = BEL
						has_war_with = LUX
					}
				}
			}
		}
		random_list = {
			40 = { ENG_wilhelm_iv_joins_effect = yes } # Eitel Friedrich agrees
			60 = { set_country_flag = ENG_wilhelm_iv_refused_throne country_event = { id = ukrework_improvements.5 days = 1 } } # Eitel Friedrich refuses
		}
	}
	option = { # ask his sister
		name = ukrework_improvements.5.c
		trigger = {
			NOT = {
				has_country_flag = ENG_victoria_refused_throne
				any_country = {
					exists = yes
					has_country_leader = {
						character = GER_victoria
						ruling_only = yes
					}
				}
			}
		}
		random_list = {
			99 = { ENG_victoria_joins_effect = yes } # Victoria agrees
			1 = { country_event = { id = ukrework_improvements.6 days = 1 } } # only if we keep the german and british throne separate
		}
	}
	option = { # Don't bother them, we want a separate monarch!
		name = ukrework_improvements.5.d
		add_political_power = -10
		add_stability = -0.005
		set_country_flag = ENG_hohenzollern_declined_flag
		custom_effect_tooltip = generic_skip_one_line_tt
		country_event = {
			id = ukrework.14
			days = 1
		}
	}
}
